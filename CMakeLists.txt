if(CONFIG_ZMK_NEMU_LINK_CCM_RADIO OR CONFIG_ZMK_NEMU_LINK_CHANNEL_SCAN)
  set_property(GLOBAL PROPERTY NEMU_LINK_CCM_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
  set_property(GLOBAL PROPERTY NEMU_LINK_CCM_BINARY_ROOT "${CMAKE_CURRENT_BINARY_DIR}")
  function(nemu_link_ccm_esb_hooks)
    get_property(root GLOBAL PROPERTY NEMU_LINK_CCM_SOURCE_ROOT)
    get_property(binary GLOBAL PROPERTY NEMU_LINK_CCM_BINARY_ROOT)
    set(sdk_source "${ZEPHYR_NRF_MODULE_DIR}/subsys/esb/esb.c")
    set(copy "${binary}/esb_ccm.c")
    set(hook_args)
    if(CONFIG_ZMK_NEMU_LINK_CHANNEL_SCAN)
      list(APPEND hook_args --scan)
      if(NOT CONFIG_ZMK_NEMU_LINK_CCM_RADIO)
        list(APPEND hook_args --scan-only)
      endif()
    endif()
    execute_process(COMMAND ${PYTHON_EXECUTABLE} "${root}/scripts/esb-ccm-source.py"
      --source "${sdk_source}" --out "${copy}" ${hook_args} RESULT_VARIABLE result)
    if(NOT result EQUAL 0)
      message(FATAL_ERROR "Nemu-link CCM SDK hooks did not match")
    endif()
    set(esb_target "..__nrf__subsys__esb")
    get_target_property(sources ${esb_target} SOURCES)
    set(replaced)
    set(count 0)
    foreach(source IN LISTS sources)
      get_filename_component(name "${source}" NAME)
      if(name STREQUAL "esb.c")
        list(APPEND replaced "${copy}")
        math(EXPR count "${count} + 1")
      else()
        list(APPEND replaced "${source}")
      endif()
    endforeach()
    if(NOT count EQUAL 1)
      message(FATAL_ERROR "Nemu-link CCM expected exactly one SDK ESB source")
    endif()
    set_property(TARGET ${esb_target} PROPERTY SOURCES "${replaced}")
    target_include_directories(${esb_target} PRIVATE "${root}/include"
      "${ZEPHYR_NRF_MODULE_DIR}/subsys/esb")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
      "${sdk_source}" "${root}/scripts/esb-ccm-source.py")
  endfunction()
  cmake_language(DEFER DIRECTORY "${APPLICATION_SOURCE_DIR}" CALL nemu_link_ccm_esb_hooks)
endif()

# ZMK listeners run in object link order. Track local positions after activity
# bookkeeping so ESB STATE snapshots include them before behavior listeners run.
# Standalone builds bubble the same position event afterward for local USB HID.
if((CONFIG_ZMK_SPLIT OR CONFIG_ZMK_NEMU_LINK_ROLE_PERIPHERAL) AND
   (CONFIG_ZMK_NEMU_LINK_MODE_SWITCH OR CONFIG_ZMK_NEMU_LINK_ROLE_PERIPHERAL))
  set_property(GLOBAL PROPERTY NEMU_LINK_INPUT_ROUTER "${CMAKE_CURRENT_SOURCE_DIR}/src/role/input_router.c")
  function(nemu_link_route_before_behaviors)
    get_property(router GLOBAL PROPERTY NEMU_LINK_INPUT_ROUTER)
    get_target_property(sources app SOURCES)
    set(routed)
    set(insertions 0)
    foreach(source IN LISTS sources)
      list(APPEND routed "${source}")
      if(source MATCHES "(^|/)activity\\.c$")
        list(APPEND routed "${router}")
        math(EXPR insertions "${insertions} + 1")
      endif()
    endforeach()
    if(NOT insertions EQUAL 1)
      message(FATAL_ERROR "ESB input routing requires one activity source before behaviors")
    endif()
    set_property(TARGET app PROPERTY SOURCES "${routed}")
  endfunction()
  cmake_language(DEFER DIRECTORY "${APPLICATION_SOURCE_DIR}" CALL nemu_link_route_before_behaviors)
endif()
